---
address: costa_rica.finance.iban
description: Costa Rican IBAN — 22 chars: "CR" + 2 ISO 7064 mod-97-10 check digits + an 18-digit BBAN (a reserved 0 + 3-digit BCCR entity code + 14-digit account). The mod-97 check digit is the real, verifiable IBAN algorithm and is computed here.
generator: tdc
---
<sequence name="bank"><gen type="template" value="costa_rica.finance.bankCode"/></sequence>
<sequence name="account"><gen type="regex" value="[0-9]{14}"/></sequence>
<sequence name="iban"><compute>
  <!-- BBAN = reserved 0 + entity code (3) + account (14) = 18 digits. -->
  <let name="bban"><concat><str v="0"/><field name="bank"/><field name="account"/></concat></let>
  <!-- ISO 13616: move the country code and a "00" placeholder to the end. -->
  <let name="rearranged"><concat><use name="bban"/><str v="CR"/><str v="00"/></concat></let>
  <!-- Letters become their base-36 value (C -> 12, R -> 27); digits stay. -->
  <let name="digits">
    <join>
      <each>
        <over><use name="rearranged"/></over>
        <do><encode as="base36"><current/></encode></do>
      </each>
    </join>
  </let>
  <!-- Horner fold, so the 24-digit value never has to exist as one number. -->
  <let name="rem">
    <reduce>
      <over><use name="digits"/></over>
      <init><int v="0"/></init>
      <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
    </reduce>
  </let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="CR"/><use name="check"/><use name="bban"/></concat></result>
</compute></sequence>
<data>${{iban}}</data>
