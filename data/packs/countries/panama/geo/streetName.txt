<sequence name="n"><gen type="number" value="1..100"/></sequence>
<sequence name="named"><gen type="template" value="panama.geo.streetNamed"/></sequence>
<sequence name="street"><compute>
  <result>
    <choose>
      <when><test><greater_than><to_number><field name="mode"/></to_number><int v="8"/></greater_than></test>
        <then><concat><str v="Avenida"/><str v=" "/><field name="n"/></concat></then></when>
      <when><test><greater_than><to_number><field name="mode"/></to_number><int v="6"/></greater_than></test>
        <then><concat><str v="Calle"/><str v=" "/><field name="n"/></concat></then></when>
      <otherwise><field name="named"/></otherwise>
    </choose>
  </result>
</compute></sequence>
<data>${{street}}</data>
