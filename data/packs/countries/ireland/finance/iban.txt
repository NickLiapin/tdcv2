---
address: ireland.finance.iban
description: Irish IBAN — 22 chars: "IE" + 2 ISO 7064 mod-97-10 check digits + BBAN (4-letter bank identifier + 6-digit National Sort Code + 8-digit account number)
generator: tdc
locale: en
---
<!-- BBAN = bank identifier (4 letters) + NSC (6 digits) + account (8 digits). -->
<sequence name="bankcode"><gen type="template" value="ireland.finance.bankCode"/></sequence>
<sequence name="nsc"><gen type="regex" value="[0-9]{6}"/></sequence>
<sequence name="account"><gen type="regex" value="[0-9]{8}"/></sequence>
<sequence name="iban"><compute>
  <let name="bban"><concat><field name="bankcode"/><field name="nsc"/><field name="account"/></concat></let>
  <!-- ISO 13616: move the country code and a "00" placeholder to the end. -->
  <let name="rearranged"><concat><use name="bban"/><str v="IE"/><str v="00"/></concat></let>
  <!-- Every char to its base-36 value (letters A->10 ... Z->35, digits stay). -->
  <let name="digits">
    <join>
      <each>
        <over><use name="rearranged"/></over>
        <do><encode as="base36"><current/></encode></do>
      </each>
    </join>
  </let>
  <!-- Horner fold mod 97, so the long value never has to exist as one number. -->
  <let name="rem">
    <reduce>
      <over><use name="digits"/></over>
      <init><int v="0"/></init>
      <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
    </reduce>
  </let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="IE"/><use name="check"/><use name="bban"/></concat></result>
</compute></sequence>
<data>${{iban}}</data>
