---
address: croatia.finance.iban
description: Croatian IBAN — 21 chars: HR + 2 ISO 7064 MOD-97 check digits + a 7-digit bank code (vodeći broj banke, VBDI) + a 10-digit account number. The bank codes drawn here are real VBDIs, each confirmed against a published Croatian IBAN. Accounts denominated in euro since 1 January 2023, when Croatia left the kuna; the IBAN itself did not change.
generator: tdc

<sequence name="account"><gen type="regex" value="[0-9]{10}"/></sequence>
<sequence name="iban"><compute>
  <let name="bban"><concat><field name="bank"/><field name="account"/></concat></let>
  <let name="rearranged"><concat><use name="bban"/><str v="HR"/><str v="00"/></concat></let>
  <let name="digits"><join><each>
    <over><use name="rearranged"/></over>
    <do><encode as="base36"><current/></encode></do>
  </each></join></let>
  <let name="rem"><reduce>
    <over><use name="digits"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="HR"/><use name="check"/><use name="bban"/></concat></result>
</compute></sequence>
