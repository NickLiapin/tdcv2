address: united_kingdom.docs.nhs
description: UK NHS number — 10 digits: a 9-digit body plus a mod-11 check digit. The body is weighted 10,9,8,7,6,5,4,3,2; check = 11 − (sum mod 11), with 11 mapped to 0. A body whose check would be 10 is not a valid NHS number, so in that single case the last body digit is nudged so the emitted number always validates.
generator: tdc
---
<sequence name="raw"><gen type="regex" value="[0-9]{9}"/></sequence>
<sequence name="nhs"><compute>
  <!-- Weighted sum of the raw 9-digit body: weight = 10 - position, i.e. 10..2. -->
  <let name="sumRaw"><reduce>
    <over><field name="raw"/></over>
    <init><int v="0"/></init>
    <do><add><acc/>
      <multiply><current/><subtract><int v="10"/><current_index/></subtract></multiply>
    </add></do>
  </reduce></let>
  <let name="rRaw"><mod><use name="sumRaw"/><int v="11"/></mod></let>
  <!-- check = 11 - (sum mod 11) is a single digit 0-9 EXCEPT when the remainder
       is 1 (check would be 10, never issued). In that one case bump the last body
       digit by 1 (mod 10): that shifts the remainder to 3 or 5, off both 1 and 0,
       so every emitted number has a valid single-digit check. -->
  <let name="last"><to_number><slice from="8" to="9"><field name="raw"/></slice></to_number></let>
  <let name="bumped"><mod><add><use name="last"/><int v="1"/></add><int v="10"/></mod></let>
  <let name="body"><choose>
    <when><test><equals><use name="rRaw"/><int v="1"/></equals></test>
      <then><concat><slice from="0" to="8"><field name="raw"/></slice><use name="bumped"/></concat></then></when>
    <otherwise><field name="raw"/></otherwise>
  </choose></let>
  <let name="sum"><reduce>
    <over><use name="body"/></over>
    <init><int v="0"/></init>
    <do><add><acc/>
      <multiply><current/><subtract><int v="10"/><current_index/></subtract></multiply>
    </add></do>
  </reduce></let>
  <let name="r"><mod><use name="sum"/><int v="11"/></mod></let>
  <let name="check"><choose>
    <when><test><equals><use name="r"/><int v="0"/></equals></test><then><int v="0"/></then></when>
    <otherwise><subtract><int v="11"/><use name="r"/></subtract></otherwise>
  </choose></let>
  <result><concat><use name="body"/><use name="check"/></concat></result>
</compute></sequence>
