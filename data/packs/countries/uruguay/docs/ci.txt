description: Uruguay Cédula de Identidad — a 7-digit number plus a check digit computed with the weights 2,9,8,7,6,3,4 as (10 - weighted sum mod 10) mod 10. Emitted bare (12345672); the DNIC prints it as 1.234.567-2. Serials in circulation run from about 1.000.000 upward, so the leading digit is never 0.
generator: tdc

<!-- Shorter historic cédulas are padded to seven digits before the weights are
     applied; every serial the DNIC still issues is already seven digits. -->
<sequence name="body"><gen type="regex" value="[1-9][0-9]{6}"/></sequence>
<sequence name="ci"><compute>
  <let name="padded"><pad width="7" fill="0"><field name="body"/></pad></let>
  <let name="sum">
    <reduce>
      <over><var name="padded"/></over>
      <init><int v="0"/></init>
      <do><add><acc/>
        <multiply><current/>
          <at><in><list v="2,9,8,7,6,3,4"/></in><index><current_index/></index></at>
        </multiply>
      </add></do>
    </reduce>
  </let>
  <!-- (10 - sum mod 10) mod 10, so a sum ending in 0 yields the digit 0.
       Published example: 1.234.567 sums to 148 and checks to 2. -->
  <let name="check"><mod><subtract><int v="10"/><mod><var name="sum"/><int v="10"/></mod></subtract><int v="10"/></mod></let>
  <result><concat><var name="padded"/><var name="check"/></concat></result>
</compute></sequence>
