---
address: chile.docs.runFormatted
description: Chilean RUN/RUT written the way Chile writes it — thousands separated by dots and the check character after a hyphen, e.g. 12.345.678-5 or 6.376.104-K. Same módulo 11 check as chile.docs.run
generator: tdc
---
<!-- The body is built in its three printed groups so the dots fall in the
     right place for both 7-digit (d.ddd.ddd) and 8-digit (dd.ddd.ddd) RUNs. -->
<sequence name="head"><gen type="regex" value="[1-9][0-9]?"/></sequence>
<sequence name="mid"><gen type="regex" value="[0-9]{3}"/></sequence>
<sequence name="tail"><gen type="regex" value="[0-9]{3}"/></sequence>
<sequence name="body"><compute>
  <result><concat><field name="head"/><field name="mid"/><field name="tail"/></concat></result>
</compute></sequence>
<sequence name="rut"><compute>
  <let name="len"><length><field name="body"/></length></let>
  <!-- Módulo 11: walking right to left the weights cycle 2,3,4,5,6,7,2,3,...
       Index i from the left sits (len-1-i) places from the right, so its
       weight is 2 + ((len-1-i) mod 6). -->
  <let name="sum">
    <reduce>
      <over><field name="body"/></over>
      <init><int v="0"/></init>
      <do><add><acc/>
        <multiply><current/>
          <add><int v="2"/>
            <mod>
              <subtract><var name="len"/><int v="1"/><current_index/></subtract>
              <int v="6"/>
            </mod>
          </add>
        </multiply>
      </add></do>
    </reduce>
  </let>
  <!-- 11 - (sum mod 11), folded so 11 becomes 0; a remainder of 1 leaves 10,
       which prints as K. Roughly one RUT in eleven ends in K. -->
  <let name="c"><mod><subtract><int v="11"/><mod><var name="sum"/><int v="11"/></mod></subtract><int v="11"/></mod></let>
  <result>
    <choose>
      <when><test><equals><var name="c"/><int v="10"/></equals></test><then><str v="K"/></then></when>
      <otherwise><var name="c"/></otherwise>
    </choose>
  </result>
</compute></sequence>
<data>${{head}}.${{mid}}.${{tail}}-${{rut}}</data>
