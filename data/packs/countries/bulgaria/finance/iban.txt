---
address: bulgaria.finance.iban
description: Bulgarian IBAN — 22 characters: BG + 2 ISO 7064 MOD-97-10 check digits + the bank's 4-letter BIC stem + a 4-digit branch (БАЕ) code + a 2-digit account-type code + an 8-character account number. The BIC stems drawn here belong to banks that hold a current Bulgarian licence; the account-type code is drawn freely because the BNB's code table is not published as a fixed list. Bulgaria adopted the euro on 1 January 2026 at 1.95583 BGN to the euro, replacing the лев; IBANs were not reissued, so the format is unchanged by the changeover.
generator: tdc
---
<sequence name="bank"><gen type="text" value="UNCR,STSA,UBBS,BPBI,FINV,CECB,BUIN,IORT,IABG,SOMB,TEXI,NASB,DEMI,KLBB,PRCB,CREX,BGUS,BNBG"/></sequence>
<sequence name="branch"><gen type="regex" value="[0-9]{4}"/></sequence>
<sequence name="acctype"><gen type="regex" value="[0-9]{2}"/></sequence>
<sequence name="account"><gen type="regex" value="[0-9]{8}"/></sequence>
<sequence name="iban"><compute>
  <let name="bban"><concat><field name="bank"/><field name="branch"/><field name="acctype"/><field name="account"/></concat></let>
  <!-- MOD-97: move BG00 to the back, letter -> two digits (A=10 … Z=35), remainder of 97. -->
  <let name="rearranged"><concat><use name="bban"/><str v="BG"/><str v="00"/></concat></let>
  <let name="digits"><join><each>
    <over><use name="rearranged"/></over>
    <do><encode as="base36"><current/></encode></do>
  </each></join></let>
  <let name="rem"><reduce>
    <over><use name="digits"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="BG"/><use name="check"/><use name="bban"/></concat></result>
</compute></sequence>
<data>${{iban}}</data>
