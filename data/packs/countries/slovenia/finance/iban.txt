address: slovenia.finance.iban
description: Slovenian IBAN — 19 characters, one of the two shortest in the euro area alongside Belgium's 16 and Estonia's 20, and eight shorter than a Maltese one. The BBAN is 15 digits: a 2-digit bank code, a 3-digit branch, an 8-digit account and 2 national check digits. Those national digits are themselves ISO 7064 MOD 97-10 over the preceding 13, and that has a consequence worth knowing: because the whole BBAN then leaves remainder 1 mod 97, every correctly formed Slovenian IBAN begins SI56. The 56 is still COMPUTED here rather than written in, so a wrong national check would show up immediately as a different pair. Bank codes drawn here are the live ones from the Banka Slovenije register: 01 Banka Slovenije, 02 and 30 NLB, 04 OTP banka, 07 Gorenjska banka, 10 Banka Intesa Sanpaolo, 19 Deželna banka Slovenije, 29 UniCredit, 33 Addiko, 34 Sparkasse, 35 BKS, 38 SID banka, 60 Hranilnica LON, 61 Delavska hranilnica, 64 Primorska hranilnica Vipava. Code 03 was SKB and is gone: SKB merged into OTP banka in 2024, as Abanka (05) and Banka Celje (06) had merged into Nova KBM before it. Accounts are in euro and have been since 1 January 2007. Verified against the published IBANs SI56263300012039086, SI56040000279259061, SI56192001234567892 and SI56191000000123438.
generator: tdc

<sequence name="bank"><gen type="text" value="01,02,04,07,10,19,29,30,33,34,35,38,60,61,64"/></sequence>
<sequence name="branch"><gen type="regex" value="[0-9]{3}"/></sequence>
<sequence name="account"><gen type="regex" value="[0-9]{8}"/></sequence>
<sequence name="iban"><compute>
  <let name="n13"><concat><field name="bank"/><field name="branch"/><field name="account"/></concat></let>
  <!-- National check: ISO 7064 MOD 97-10 over the 13 digits, so that BBAN mod 97 == 1. -->
  <let name="r13"><reduce>
    <over><use name="n13"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="nat"><pad width="2" fill="0">
    <mod><subtract><int v="98"/><mod><multiply><use name="r13"/><int v="3"/></multiply><int v="97"/></mod></subtract><int v="97"/></mod>
  </pad></let>
  <let name="bban"><concat><use name="n13"/><use name="nat"/></concat></let>
  <!-- IBAN check: the usual MOD-97 over BBAN + SI + 00. It always lands on 56. -->
  <let name="rearranged"><concat><use name="bban"/><str v="SI"/><str v="00"/></concat></let>
  <let name="digits"><join><each>
    <over><use name="rearranged"/></over>
    <do><encode as="base36"><current/></encode></do>
  </each></join></let>
  <let name="rem"><reduce>
    <over><use name="digits"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="SI"/><use name="check"/><use name="bban"/></concat></result>
</compute></sequence>
