---
address: finland.finance.iban
description: Finnish IBAN — 18 chars: FI + 2 ISO 7064 check digits + a 6-digit bank code + an 8-digit account number. The ISO 7064 check digits are computed; the bank code and account body are structural, not drawn from real issued ranges.
generator: tdc
locale: fi
---
<sequence name="bankcode"><gen type="regex" value="[1-8][0-9]{5}"/></sequence>
<sequence name="account"><gen type="regex" value="[0-9]{8}"/></sequence>
<sequence name="iban"><compute>
  <let name="bban"><concat><field name="bankcode"/><field name="account"/></concat></let>
  <let name="rearranged"><concat><use name="bban"/><str v="FI"/><str v="00"/></concat></let>
  <let name="digits"><join><each>
    <over><use name="rearranged"/></over>
    <do><encode as="base36"><current/></encode></do>
  </each></join></let>
  <let name="rem"><reduce>
    <over><use name="digits"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="FI"/><use name="check"/><use name="bban"/></concat></result>
</compute></sequence>
<data>${{iban}}</data>
