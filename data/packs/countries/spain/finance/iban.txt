---
address: spain.finance.iban
description: Spanish IBAN — 24 chars: "ES" + 2 ISO 7064 mod-97-10 check digits + the 20-digit CCC (entity, oficina, 2 national control digits, account). The national control digits and the IBAN check digits are two different algorithms; both are computed here.
generator: tdc
---
<!-- The BBAN of a Spanish IBAN is exactly the CCC, national control digits
     included — so it is built by the CCC generator and only the ISO check
     digits are added on top. -->
<sequence name="bban"><gen type="template" value="spain.finance.ccc"/></sequence>
<sequence name="iban"><compute>
  <!-- ISO 13616: move the country code and a "00" placeholder to the end. -->
  <let name="rearranged"><concat><field name="bban"/><str v="ES"/><str v="00"/></concat></let>
  <!-- Letters become their base-36 value (E -> 14, S -> 28); digits stay. -->
  <let name="digits">
    <join>
      <each>
        <over><use name="rearranged"/></over>
        <do><encode as="base36"><current/></encode></do>
      </each>
    </join>
  </let>
  <!-- Horner fold, so the 26-digit value never has to exist as one number. -->
  <let name="rem">
    <reduce>
      <over><use name="digits"/></over>
      <init><int v="0"/></init>
      <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
    </reduce>
  </let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="ES"/><use name="check"/><field name="bban"/></concat></result>
</compute></sequence>
<data>${{iban}}</data>
