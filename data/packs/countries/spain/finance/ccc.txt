      </multiply>
    </add></do>

  <!-- d = 11 - (sum mod 11); the two overflow cases collapse: 11 -> 0, 10 -> 1. -->

    <otherwise><subtract><int v="11"/><mod><use name="sumTail"/><int v="11"/></mod></subtract></otherwise>
  </choose></let>
  <result><concat>
    <field name="entidad"/><field name="oficina"/>
