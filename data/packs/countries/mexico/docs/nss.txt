<sequence name="office"><gen type="number" value="1..99" length="2" first_zero="true"/></sequence>
<sequence name="regyear"><gen type="regex" value="[0-9]{2}"/></sequence>
<sequence name="birthyear"><gen type="regex" value="[0-9]{2}"/></sequence>
<sequence name="serial"><gen type="regex" value="[0-9]{4}"/></sequence>
<sequence name="nss"><compute>
  <!-- 10-digit payload; the 11th digit is the Luhn check. -->
  <let name="base"><concat>
    <field name="office"/><field name="regyear"/><field name="birthyear"/><field name="serial"/>
  </concat></let>
  <!-- Luhn: counting from the right of the finished number, every second digit
       doubles. With the check digit appended, that is the ODD indices of the
       10-digit payload (1,3,5,7,9). Doubled values over 9 lose 9. -->
  <let name="sum"><reduce>
    <over><var name="base"/></over>
    <init><int v="0"/></init>
    <do><add><acc/>
      <choose>
        <when><test><equals><mod><current_index/><int v="2"/></mod><int v="1"/></equals></test>
          <then>
            <choose>
              <when><test><greater_than><current/><int v="4"/></greater_than></test>
                <then><subtract><multiply><current/><int v="2"/></multiply><int v="9"/></subtract></then></when>
              <otherwise><multiply><current/><int v="2"/></multiply></otherwise>
            </choose>
          </then></when>
        <otherwise><current/></otherwise>
      </choose>
    </add></do>
  </reduce></let>
  <let name="check">
    <mod><subtract><int v="10"/><mod><var name="sum"/><int v="10"/></mod></subtract><int v="10"/></mod>
  </let>
  <result><concat><var name="base"/><var name="check"/></concat></result>
</compute></sequence>
