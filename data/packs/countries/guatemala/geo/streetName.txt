<sequence name="street"><compute>
  <!-- A quarter of the streets are named calzadas; the rest are numbered,

  <result>
    <choose>
      <when><test><less_than><to_number><field name="mode"/></to_number><int v="4"/></less_than></test>

        <then><concat><field name="n"/><str v="a Avenida"/></concat></then></when>
      <when><test><less_than><to_number><field name="mode"/></to_number><int v="11"/></less_than></test>
        <then><concat><field name="n"/><str v=" Calle"/></concat></then></when>
      <otherwise><concat><field name="diag"/><str v=" "/><field name="n"/></concat></otherwise>
    </choose>
  </result>
</compute></sequence>
