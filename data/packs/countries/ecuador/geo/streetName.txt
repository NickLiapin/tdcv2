---
address: ecuador.geo.streetName
description: Ecuadorian street name with its street type — mostly named after próceres, patriotic dates and provinces (Av. Amazonas, Calle Sucre); a minority are pasajes or numbered ordinals
generator: tdc
---
<sequence name="mode"><gen type="number" value="1..10"/></sequence>
<sequence name="kind"><gen type="text" value="Calle,Calle,Calle,Avenida,Avenida,Avenida,Pasaje,Pasaje,Callejón,Malecón"/></sequence>
<!-- Ecuadorian addresses often carry an ordinal name (Calle Segunda, Pasaje Quinto). -->
<sequence name="ordinal"><gen type="text" value="Primera,Segunda,Tercera,Cuarta,Quinta,Sexta,Séptima,Octava,Novena,Décima"/></sequence>
<sequence name="named"><gen type="template" value="ecuador.geo.streetNamed"/></sequence>
<sequence name="street"><compute>
  <!-- One street in five is an ordinal calle/pasaje; the rest are a type + name. -->
  <result>
    <choose>
      <when><test><greater_than><to_number><field name="mode"/></to_number><int v="8"/></greater_than></test>
        <then><concat><field name="kind"/><str v=" "/><field name="ordinal"/></concat></then></when>
      <otherwise><concat><field name="kind"/><str v=" "/><field name="named"/></concat></otherwise>
    </choose>
  </result>
</compute></sequence>
<data>${{street}}</data>
