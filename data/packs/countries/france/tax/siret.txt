---
address: france.tax.siret
description: French SIRET — 14 digits: a valid 9-digit SIREN + 4-digit establishment number (NIC) + Luhn check. The whole 14-digit number is Luhn-valid, and so is the embedded SIREN.

---
<!-- The SIREN is already a valid Luhn number; reuse it so the establishment
     shares a real company identifier. -->
<sequence name="siren"><gen type="template" value="france.tax.siren"/></sequence>
<sequence name="nic"><gen type="regex" value="[0-9]{4}"/></sequence>
<sequence name="siret"><compute>
  <let name="base"><concat><field name="siren"/><field name="nic"/></concat></let>
  <!-- Luhn over the 13-digit base; the doubled positions are those whose index
       matches the parity of (length + 1), so the appended check lands on a
       non-doubled slot. Same generalized rule as the SIREN generator. -->
  <let name="sum"><reduce>
    <over><use name="base"/></over>
    <init><int v="0"/></init>
    <do><add><acc/>
      <choose>
        <when>
          <test><equals>
            <mod><current_index/><int v="2"/></mod>
            <mod><add><length><use name="base"/></length><int v="1"/></add><int v="2"/></mod>
          </equals></test>
          <then>
            <let name="d"><multiply><current/><int v="2"/></multiply></let>
            <choose>
              <when><test><greater_than><use name="d"/><int v="9"/></greater_than></test>
                    <then><subtract><use name="d"/><int v="9"/></subtract></then></when>
              <otherwise><use name="d"/></otherwise>
            </choose>
          </then>
        </when>
        <otherwise><current/></otherwise>
      </choose>
    </add></do>
  </reduce></let>
  <let name="check"><mod><subtract><int v="10"/><mod><use name="sum"/><int v="10"/></mod></subtract><int v="10"/></mod></let>
  <result><concat><use name="base"/><use name="check"/></concat></result>
</compute></sequence>
<data>${{siret}}</data>
