<sequence name="sex"><gen type="text" value="1,2"/></sequence>
<sequence name="year"><gen type="regex" value="[0-9]{2}"/></sequence>
<sequence name="month"><gen type="number" value="1..12" length="2" first_zero="true"/></sequence>
<!-- Department of birth, kept numeric 01-95 so the whole body is digits; the
     Corsica codes 2A/2B are intentionally not emitted. -->
<sequence name="dept"><gen type="number" value="1..95" length="2" first_zero="true"/></sequence>
<sequence name="commune"><gen type="regex" value="[0-9]{3}"/></sequence>
<sequence name="serial"><gen type="regex" value="[0-9]{3}"/></sequence>
<sequence name="nir"><compute>
  <let name="body"><concat>
    <field name="sex"/><field name="year"/><field name="month"/>
    <field name="dept"/><field name="commune"/><field name="serial"/>
  </concat></let>
  <!-- Control key = 97 - (body mod 97). Horner fold keeps the 13-digit body
       from ever becoming one number; a remainder of 0 yields the key 97. -->
  <let name="rem"><reduce>
    <over><use name="body"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="ctrl"><subtract><int v="97"/><use name="rem"/></subtract></let>
  <result><concat>
    <field name="sex"/><field name="year"/><field name="month"/>
    <field name="dept"/><field name="commune"/><field name="serial"/>
    <pad width="2" fill="0"><use name="ctrl"/></pad>
  </concat></result>
</compute></sequence>
