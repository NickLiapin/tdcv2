---
address: france.finance.iban
description: French IBAN — 27 chars: FR plus 2 ISO 7064 mod-97-10 check digits plus the 23-digit BBAN (5-digit bank code plus 5-digit branch/guichet plus 11-digit account plus 2-digit RIB key). The RIB key and the IBAN check digits are two different mod-97 algorithms; both are computed here.
generator: tdc
---
<sequence name="bank"><gen type="regex" value="[0-9]{5}"/></sequence>
<sequence name="branch"><gen type="regex" value="[0-9]{5}"/></sequence>
<sequence name="account"><gen type="regex" value="[0-9]{11}"/></sequence>
<!-- BBAN = bank + branch + account + RIB key. The RIB key is 97 minus the
     mod-97 of the 21 digits with "00" appended, which is the classic
     89*bank + 15*branch + 3*account weighting expressed as a plain fold. -->
<sequence name="bban"><compute>
  <let name="body"><concat><field name="bank"/><field name="branch"/><field name="account"/></concat></let>
  <let name="ribrem"><reduce>
    <over><concat><use name="body"/><str v="00"/></concat></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="ribkey"><subtract><int v="97"/><use name="ribrem"/></subtract></let>
  <result><concat>
    <field name="bank"/><field name="branch"/><field name="account"/>
    <pad width="2" fill="0"><use name="ribkey"/></pad>
  </concat></result>
</compute></sequence>
<sequence name="iban"><compute>
  <!-- ISO 13616: move "FR00" to the end, expand letters to their base-36 value
       (F -> 15, R -> 27), Horner-fold mod 97, check = 98 - remainder. -->
  <let name="rearranged"><concat><field name="bban"/><str v="FR"/><str v="00"/></concat></let>
  <let name="digits"><join><each>
    <over><use name="rearranged"/></over>
    <do><encode as="base36"><current/></encode></do>
  </each></join></let>
  <let name="rem"><reduce>
    <over><use name="digits"/></over>
    <init><int v="0"/></init>
    <do><mod><add><multiply><acc/><int v="10"/></multiply><current/></add><int v="97"/></mod></do>
  </reduce></let>
  <let name="check"><pad width="2" fill="0"><subtract><int v="98"/><use name="rem"/></subtract></pad></let>
  <result><concat><str v="FR"/><use name="check"/><field name="bban"/></concat></result>
</compute></sequence>
<data>${{iban}}</data>
